cmake_minimum_required(VERSION 3.16)
project(fast_lio)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_BUILD_TYPE Release)

find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(pcl_conversions REQUIRED)
find_package(livox_ros_driver2 REQUIRED)
find_package(PCL REQUIRED COMPONENTS common io filters)
find_package(Eigen3 REQUIRED)
find_package(OpenMP REQUIRED)
find_package(rosidl_default_generators REQUIRED)

# ---- compile definitions the algorithm core expects from upstream CMakeLists:
#      ROOT_DIR (log/PCD output path) and the OpenMP knobs MP_EN / MP_PROC_NUM.
#      Missing these is a build error, not a runtime one -- found by compiling.
add_definitions(-DROOT_DIR="${CMAKE_CURRENT_SOURCE_DIR}/")
include(ProcessorCount)
ProcessorCount(NPROC)
if(NPROC GREATER 3)
  add_definitions(-DMP_EN)
  add_definitions(-DMP_PROC_NUM=3)
else()
  add_definitions(-DMP_PROC_NUM=1)
endif()

# ---- custom messages (ROS1: add_message_files + generate_messages) ----
rosidl_generate_interfaces(${PROJECT_NAME}
  "msg/VIO.msg"
  "msg/Pose6D.msg"
  DEPENDENCIES std_msgs geometry_msgs
)

add_executable(fastlio_mapping
  src/main.cpp
  src/ros2node.cpp
  # algorithm core: generated by extract_core.py from the upstream ROS1 source
  src/fastlio_core.cpp
  src/preprocess.cpp
  include/ikd-Tree/ikd_Tree.cpp
)

# the generated message headers are a build dependency of the executable
add_dependencies(fastlio_mapping ${PROJECT_NAME}__cpp)

# link against the C++ typesupport library produced above
rosidl_get_typesupport_target(cpp_typesupport_target ${PROJECT_NAME} rosidl_typesupport_cpp)
target_link_libraries(fastlio_mapping "${cpp_typesupport_target}")

ament_target_dependencies(fastlio_mapping
  rclcpp std_msgs sensor_msgs nav_msgs geometry_msgs tf2_ros pcl_conversions livox_ros_driver2
)

target_include_directories(fastlio_mapping PRIVATE
  include
  src
  ${PCL_INCLUDE_DIRS}
  ${EIGEN3_INCLUDE_DIR}
)
target_link_libraries(fastlio_mapping ${PCL_LIBRARIES} OpenMP::OpenMP_CXX)

install(TARGETS fastlio_mapping DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config rviz DESTINATION share/${PROJECT_NAME})

ament_export_dependencies(rosidl_default_runtime)
ament_package()
